                if (localHtmlFile.exists()) localHtmlFile.delete()

        }

    private fun loadOfflineCluster() {

        }
    }

    private fun pushInstalledAppsToWeb() {
        thread {
            try {
                val jsonStr = queryInstalledAppsJsonString()
                val escaped = JSONObject.quote(jsonStr)
                runOnUiThread {
                    try {
                        webView.evaluateJavascript(
                            "if(window.onAndroidAppsLoaded){window.onAndroidAppsLoaded($escaped);}",
                            null
                        )
                    } catch (_: Exception) {}
                }
            } catch (_: Exception) {}
        }
    }

    private fun queryInstalledAppsJsonString(): String {
        val pm = packageManager
        val seen = HashSet<String>()
        val arr = JSONArray()

        val categories = listOf(Intent.CATEGORY_LAUNCHER, Intent.CATEGORY_LEANBACK_LAUNCHER)
        for (cat in categories) {
            val intent = Intent(Intent.ACTION_MAIN, null).apply { addCategory(cat) }
            val list = try {
                pm.queryIntentActivities(intent, 0)
            } catch (_: Exception) {
                emptyList()
            }
            for (ri in list) {
                val pkg = ri.activityInfo?.packageName ?: continue
                if (pkg == packageName || seen.contains(pkg)) continue
                seen.add(pkg)
                val label = try {
                    ri.loadLabel(pm)?.toString() ?: pkg
                } catch (_: Exception) {
                    pkg
                }
                val obj = JSONObject()
                obj.put("name", label)
                obj.put("pkg", pkg)
                arr.put(obj)
            }
        }
        return arr.toString()
    }

    private fun registerWifiAutoUpdateCallback() {
        try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                val cm = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
                val req = NetworkRequest.Builder()
                    .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                    .build()
                cm.registerNetworkCallback(req, object : ConnectivityManager.NetworkCallback() {
                    override fun onAvailable(network: Network) {
                        syncFromGitHubInBackground(forceReload = false, showFeedback = true)
                    }
                })
            }
        } catch (_: Exception) {}
    }

    private fun syncFromGitHubInBackground(forceReload: Boolean, showFeedback: Boolean) {
        thread {
            try {
                val sep = if (githubRawUrl.contains("?")) "&" else "?"
                val targetUrl = githubRawUrl + sep + "t=" + System.currentTimeMillis()
                val conn = (URL(targetUrl).openConnection() as HttpURLConnection).apply {
                    connectTimeout = 6000
                    readTimeout = 8000
                    requestMethod = "GET"
                    useCaches = false
                    setRequestProperty("Cache-Control", "no-cache, no-store, must-revalidate")
                    setRequestProperty("Pragma", "no-cache")
                }
                if (conn.responseCode == 200) {
                    val newHtml = conn.inputStream.bufferedReader(Charsets.UTF_8).readText()
                    if (newHtml.contains("<!DOCTYPE html>", ignoreCase = true) &&
                        newHtml.contains("CHEVROLET_COCKPIT_V4")
                    ) {
                        val oldHtml = if (localHtmlFile.exists()) localHtmlFile.readText() else ""
                        if (newHtml != oldHtml || forceReload) {
                            localHtmlFile.writeText(newHtml, Charsets.UTF_8)
                            runOnUiThread {
                                webView.loadUrl("file://${localHtmlFile.absolutePath}")
                                if (showFeedback) {
                                    Toast.makeText(
                                        this@MainActivity,
                                        "Sistema actualizado desde GitHub",
                                        Toast.LENGTH_SHORT
                                    ).show()
                                }
                            }
                        } else if (showFeedback) {
                            runOnUiThread {
                                Toast.makeText(
                                    this@MainActivity,
                                    "El sistema ya está en la última versión",
                                    Toast.LENGTH_SHORT
                                ).show()
                            }
                        }
                    }
                }
            } catch (_: Exception) {}
        }
    }

    inner class CockpitBridge {

        @JavascriptInterface
        fun getInstalledAppsJson(): String {
            return try {
                queryInstalledAppsJsonString()
            } catch (_: Exception) {
                "[]"
            }
        }

        @JavascriptInterface
        fun launchApp(pkgName: String?): Boolean {
            return launchAppInternal(pkgName, splitScreen = false)
        }

        @JavascriptInterface
        fun launchAppSplitScreen(pkgName: String?): Boolean {
            return launchAppInternal(pkgName, splitScreen = true)
        }

        private fun launchAppInternal(pkgName: String?, splitScreen: Boolean): Boolean {
            if (pkgName.isNullOrBlank()) return false
            val cleanPkg = pkgName.trim()
            return try {
                var launchIntent = packageManager.getLaunchIntentForPackage(cleanPkg)
                if (launchIntent == null && Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                    launchIntent = packageManager.getLeanbackLaunchIntentForPackage(cleanPkg)
                }
                if (launchIntent != null) {
                    launchIntent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                    if (splitScreen && Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
                        launchIntent.addFlags(Intent.FLAG_ACTIVITY_LAUNCH_ADJACENT or Intent.FLAG_ACTIVITY_MULTIPLE_TASK)
                    }
                    startActivity(launchIntent)
                    true
                } else {
                    runOnUiThread {
                        Toast.makeText(
                            this@MainActivity,
                            "Aplicación no instalada ($cleanPkg)",
                            Toast.LENGTH_SHORT
                        ).show()
                    }
                    false
                }
            } catch (_: Exception) {
                false
            }
        }

        @JavascriptInterface
        fun openIntentAction(actionType: String?) {
            val act = actionType?.trim()?.lowercase() ?: "settings"
            try {
                val intent = when (act) {
                    "wifi" -> Intent(Settings.ACTION_WIFI_SETTINGS)
                    "bluetooth" -> Intent(Settings.ACTION_BLUETOOTH_SETTINGS)
                    "sound" -> Intent(Settings.ACTION_SOUND_SETTINGS)
                    "display" -> Intent(Settings.ACTION_DISPLAY_SETTINGS)
                    "apps" -> Intent(Settings.ACTION_APPLICATION_SETTINGS)
                    "storage" -> Intent(Settings.ACTION_INTERNAL_STORAGE_SETTINGS)
                    "location" -> Intent(Settings.ACTION_LOCATION_SOURCE_SETTINGS)
                    "maps" -> Intent(Intent.ACTION_VIEW, Uri.parse("geo:0,0?q="))
                    else -> Intent(Settings.ACTION_SETTINGS)
                }
                intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                startActivity(intent)
            } catch (_: Exception) {
                try {
                    startActivity(Intent(Settings.ACTION_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
                } catch (_: Exception) {}
            }
        }

        @JavascriptInterface
        fun openWifiSettings() {
            openIntentAction("wifi")
        }

        @JavascriptInterface
        fun openSystemSettings() {
            openIntentAction("settings")
        }

        @JavascriptInterface
        fun mediaKeyControl(command: String?) {
            try {
                val audioManager = getSystemService(Context.AUDIO_SERVICE) as AudioManager
                when (command?.lowercase()) {
                    "vol_up" -> audioManager.adjustStreamVolume(
                        AudioManager.STREAM_MUSIC,
                        AudioManager.ADJUST_RAISE,
                        AudioManager.FLAG_SHOW_UI
                    )
                    "vol_down" -> audioManager.adjustStreamVolume(
                        AudioManager.STREAM_MUSIC,
                        AudioManager.ADJUST_LOWER,
                        AudioManager.FLAG_SHOW_UI
                    )
                    "mute" -> audioManager.adjustStreamVolume(
                        AudioManager.STREAM_MUSIC,
                        AudioManager.ADJUST_TOGGLE_MUTE,
                        AudioManager.FLAG_SHOW_UI
                    )
                    "play_pause" -> sendMediaKey(audioManager, KeyEvent.KEYCODE_MEDIA_PLAY_PAUSE)
                    "next" -> sendMediaKey(audioManager, KeyEvent.KEYCODE_MEDIA_NEXT)
                    "prev" -> sendMediaKey(audioManager, KeyEvent.KEYCODE_MEDIA_PREVIOUS)
                }
            } catch (_: Exception) {}
        }

        private fun sendMediaKey(am: AudioManager, keyCode: Int) {
            am.dispatchMediaKeyEvent(KeyEvent(KeyEvent.ACTION_DOWN, keyCode))
            am.dispatchMediaKeyEvent(KeyEvent(KeyEvent.ACTION_UP, keyCode))
        }

        @JavascriptInterface
        fun checkForGitHubUpdate(customUrl: String?) {
            if (!customUrl.isNullOrBlank() && customUrl.startsWith("http")) {
                githubRawUrl = customUrl.trim()
                prefs.edit().putString("github_ota_url", githubRawUrl).apply()
            }
            syncFromGitHubInBackground(forceReload = true, showFeedback = true)
        }

        @JavascriptInterface
        fun resetToFactoryAsset() {
            try {
                if (localHtmlFile.exists()) localHtmlFile.delete()
                runOnUiThread {
                    webView.loadUrl("file:///android_asset/index.html")
                }
            } catch (_: Exception) {}
        }
    }

    private fun hideSystemBars() {
        try {
            @Suppress("DEPRECATION")
            window.decorView.systemUiVisibility = (
                View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY
                    or View.SYSTEM_UI_FLAG_FULLSCREEN
                    or View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
                    or View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
                    or View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
                    or View.SYSTEM_UI_FLAG_LAYOUT_STABLE
                )
        } catch (_: Exception) {}
    }

    @Deprecated("Deprecated in Java")
    override fun onBackPressed() {
        try {
            webView.evaluateJavascript("if(window.onHardwareBack){window.onHardwareBack();}", null)
        } catch (_: Exception) {}
    }

// TELEMETRIA CHEVROLET CORSA -> USB SERIAL (115200 BAUDIOS)
#include <Arduino.h>

#define PIN_RPM_PULSE     25
#define PIN_VSS_PULSE     26
#define PIN_FUEL_ADC      34
#define PIN_TEMP_ADC      35
#define PIN_BATT_ADC      32

#define PIN_TURN_L        13
#define PIN_TURN_R        12
#define PIN_LOW_BEAM      14
#define PIN_HIGH_BEAM     27
#define PIN_FOG_POS       18
#define PIN_FOOT_BRAKE    33
#define PIN_HANDBRAKE     15
#define PIN_CHECK_ENG     4
#define PIN_OIL_WARN      16
#define PIN_BATT_WARN     17

volatile unsigned long rpmPulses = 0;
volatile unsigned long vssPulses = 0;
unsigned long lastSampleMs = 0;

void IRAM_ATTR onRpmPulse() { rpmPulses++; }
void IRAM_ATTR onVssPulse() { vssPulses++; }

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RPM_PULSE, INPUT_PULLUP);
  pinMode(PIN_VSS_PULSE, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_RPM_PULSE), onRpmPulse, FALLING);
  attachInterrupt(digitalPinToInterrupt(PIN_VSS_PULSE), onVssPulse, FALLING);

  int inputs[] = {PIN_TURN_L, PIN_TURN_R, PIN_LOW_BEAM, PIN_HIGH_BEAM, PIN_FOG_POS,
                  PIN_FOOT_BRAKE, PIN_HANDBRAKE, PIN_CHECK_ENG, PIN_OIL_WARN, PIN_BATT_WARN};
  for (int i = 0; i < 10; i++) pinMode(inputs[i], INPUT_PULLUP);
}

void loop() {
  unsigned long now = millis();
  if (now - lastSampleMs >= 30) {
    float dt = (now - lastSampleMs) / 1000.0;
    lastSampleMs = now;

    noInterrupts();
    unsigned long rCount = rpmPulses; rpmPulses = 0;
    unsigned long vCount = vssPulses; vssPulses = 0;
    interrupts();

    int rpm = (int)((rCount / dt) * 30.0);
    int speedKmh = (int)((vCount / dt) * 0.85);
    int fuelPct = constrain(map(analogRead(PIN_FUEL_ADC), 400, 3600, 0, 100), 0, 100);
    int coolantC = constrain(map(analogRead(PIN_TEMP_ADC), 500, 3400, 50, 120), 50, 125);
    float battV = (analogRead(PIN_BATT_ADC) / 4095.0) * 18.5;

    Serial.printf(
      "{\"rpm\":%d,\"spd\":%d,\"fuel\":%d,\"temp\":%d,\"volt\":%.1f,"
      "\"turnL\":%d,\"turnR\":%d,\"lowBeam\":%d,\"highBeam\":%d,\"fog\":%d,"
      "\"brake\":%d,\"handbrake\":%d,\"check\":%d,\"oil\":%d,\"batt\":%d}\n",
      rpm, speedKmh, fuelPct, coolantC, battV,
      !digitalRead(PIN_TURN_L), !digitalRead(PIN_TURN_R),
      !digitalRead(PIN_LOW_BEAM), !digitalRead(PIN_HIGH_BEAM), !digitalRead(PIN_FOG_POS),
      !digitalRead(PIN_FOOT_BRAKE), !digitalRead(PIN_HANDBRAKE),
      !digitalRead(PIN_CHECK_ENG), !digitalRead(PIN_OIL_WARN), !digitalRead(PIN_BATT_WARN)
    );
  }
}
