Paso 1: Dejar la TV Box ultra liviana y sin animaciones

Conectá tu PC y la TV Box al mismo Wi-Fi (o por cable USB) y ejecutá estos comandos por adb. Esto apaga las animaciones de Android, elimina el bloqueo de pantalla, desactiva las apps pesadas de fábrica y convierte tu app de Android Studio en el único Launcher de inicio: 

# 1. Conectar a la TV Box (reemplazá por la IP de tu TV Box o usá USB)
adb connect 192.168.100.10:5555

# 2. Velocidad extrema: apagar todas las animaciones y el bootanimation
adb shell settings put global window_animation_scale 0
adb shell settings put global transition_animation_scale 0
adb shell settings put global animator_duration_scale 0
adb shell setprop debug.sf.nobootanimation 1

# 3. Quitar pantalla de bloqueo y mantener pantalla encendida con contacto
adb shell locksettings set-disabled true
adb shell settings put global stay_on_while_plugged_in 7

# 4. Ocultar barras de navegación y estado permanentemente (Modo Inmersivo)
adb shell settings put global policy_control immersive.full=*

# 5. Desactivar bloatware típico de TV Box para liberar memoria RAM
adb shell pm disable-user --user 0 com.google.android.youtube.tv
adb shell pm disable-user --user 0 com.netflix.ninja
adb shell pm disable-user --user 0 com.android.vending
adb shell pm disable-user --user 0 com.google.android.leanbacklauncher

# 6. Fijar nuestra app de Android Studio como el Launcher Principal (Home)
adb shell cmd package set-home-activity com.corsa.cockpitos/.MainActivity

Paso 2: Cómo funciona Offline en el auto y se actualiza desde GitHub

En Android Studio, guardás este HTML por primera vez en app/src/main/assets/index.html. Al arrancar el auto sin internet (Offline), la app carga instantáneamente la última versión guardada en la memoria interna de la TV Box (cockpit_latest.html) o el archivo de assets en 0.1 segundos. Cuando la TV Box detecta Wi-Fi (en tu casa o con el hotspot del celular), descarga en segundo plano el index.html nuevo desde tu repositorio de GitHub, lo guarda en la memoria interna para uso offline y actualiza la pantalla al instante: 

package com.corsa.cockpitos

import android.annotation.SuppressLint
import android.content.Context
import android.content.Intent
import android.net.ConnectivityManager
import android.net.Network
import android.net.NetworkCapabilities
import android.net.NetworkRequest
import android.os.Bundle
import android.provider.Settings
import android.view.View
import android.view.WindowManager
import android.webkit.JavascriptInterface
import android.webkit.WebSettings
import android.webkit.WebView
import android.webkit.WebViewClient
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import java.io.File
import java.net.HttpURLConnection
import java.net.URL
import kotlin.concurrent.thread

class MainActivity : AppCompatActivity() {

    private lateinit var webView: WebView
    private val localHtmlFile by lazy { File(filesDir, "cockpit_latest.html") }

    // Reemplazá por la URL Raw de tu archivo index.html en tu repositorio de GitHub:
    @Volatile
    private var githubRawUrl = "https://enfer1998.github.io/velocimetro-/"

    @SuppressLint("SetJavaScriptEnabled")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        window.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
        hideSystemBars()

        webView = WebView(this)
        setContentView(webView)

        val ws: WebSettings = webView.settings
        ws.javaScriptEnabled = true
        ws.domStorageEnabled = true
        ws.allowFileAccess = true
        ws.allowContentAccess = true
        ws.mediaPlaybackRequiresUserGesture = false
        ws.cacheMode = WebSettings.LOAD_DEFAULT

        webView.webViewClient = WebViewClient()
        webView.addJavascriptInterface(CockpitBridge(), "AndroidCockpitBridge")

        // 1. CARGA INSTANTÁNEA OFFLINE-FIRST (0.1s sin internet)
        loadOfflineCluster()

        // 2. ESCUCHAR CONEXIÓN WI-FI EN TIEMPO REAL:
        // Apenas activás el Wi-Fi en Ajustes y se conecta, descarga solo el HTML nuevo de GitHub!
        registerWifiAutoUpdateCallback()
    }

    override fun onResume() {
        super.onResume()
        hideSystemBars()
        // Al volver de la pantalla de Ajustes de Wi-Fi al Launcher, verifica si hay actualización
        syncFromGitHubInBackground(forceReload = false, showFeedback = false)
    }

    private fun loadOfflineCluster() {
        if (localHtmlFile.exists() && localHtmlFile.length() > 1000) {
            webView.loadUrl("file://${localHtmlFile.absolutePath}")
        } else {
            webView.loadUrl("file:///android_asset/index.html")
        }
    }

    private fun registerWifiAutoUpdateCallback() {
        try {
            val cm = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
            val req = NetworkRequest.Builder()
                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                .build()
            cm.registerNetworkCallback(req, object : ConnectivityManager.NetworkCallback() {
                override fun onAvailable(network: Network) {
                    // ¡Se conectó el Wi-Fi! Busca y descarga automáticamente el nuevo HTML de GitHub
                    syncFromGitHubInBackground(forceReload = false, showFeedback = true)
                }
            })
        } catch (_: Exception) {}
    }

    private fun syncFromGitHubInBackground(forceReload: Boolean, showFeedback: Boolean) {
        thread {
            try {
                val conn = (URL(githubRawUrl).openConnection() as HttpURLConnection).apply {
                    connectTimeout = 5000
                    readTimeout = 7000
                    requestMethod = "GET"
                    setRequestProperty("Cache-Control", "no-cache")
                }
                if (conn.responseCode == 200) {
                    val newHtml = conn.inputStream.bufferedReader(Charsets.UTF_8).readText()
                    if (newHtml.contains("<!DOCTYPE html>", ignoreCase = true)) {
                        val oldHtml = if (localHtmlFile.exists()) localHtmlFile.readText() else ""
                        if (newHtml != oldHtml || forceReload) {
                            localHtmlFile.writeText(newHtml, Charsets.UTF_8)
                            runOnUiThread {
                                webView.loadUrl("file://${localHtmlFile.absolutePath}")
                                if (showFeedback) {
                                    Toast.makeText(this@MainActivity, "✅ Tablero actualizado desde GitHub", Toast.LENGTH_SHORT).show()
                                }
                            }
                        }
                    }
                }
            } catch (_: Exception) {
                // Sin internet: sigue funcionando 100% offline con la versión guardada
            }
        }
    }

    inner class CockpitBridge {
        @JavascriptInterface
        fun openWifiSettings() {
            // Abre directamente el menú de Wi-Fi de Android desde el Launcher
            try {
                val intent = Intent(Settings.ACTION_WIFI_SETTINGS).apply {
                    addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                }
                startActivity(intent)
            } catch (e: Exception) {
                openSystemSettings()
            }
        }

        @JavascriptInterface
        fun openSystemSettings() {
            // Abre la Configuración General de Android (Wi-Fi, Bluetooth, Sonido, USB, Pantalla)
            try {
                val intent = Intent(Settings.ACTION_SETTINGS).apply {
                    addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                }
                startActivity(intent)
            } catch (_: Exception) {}
        }

        @JavascriptInterface
        fun checkForGitHubUpdate(customUrl: String?) {
            if (!customUrl.isNullOrBlank() && customUrl.startsWith("https://")) {
                githubRawUrl = customUrl.trim()
            }
            syncFromGitHubInBackground(forceReload = true, showFeedback = true)
        }
    }

    private fun hideSystemBars() {
        @Suppress("DEPRECATION")
        window.decorView.systemUiVisibility = (
            View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY
            or View.SYSTEM_UI_FLAG_FULLSCREEN
            or View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
            or View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
            or View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
            or View.SYSTEM_UI_FLAG_LAYOUT_STABLE
        )
    }

    override fun onBackPressed() {
        // En el Launcher bloquea el botón Atrás para no salir por accidente
    }
}

Paso 3: Configurar `AndroidManifest.xml` como Home Launcher

Al agregar las categorías CATEGORY_HOME y CATEGORY_DEFAULT en tu AndroidManifest.xml, cuando la TV Box recibe contacto ni siquiera muestra el menú de Android: bootea directo a tu tablero en orientación horizontal (landscape): 

<?xml version="1.0" encoding="utf-8"?>
<manifest xmlns:android="http://schemas.android.com/apk/res/android"
    package="com.corsa.cockpitos">

    <!-- Permiso para buscar actualizaciones OTA en GitHub cuando hay Wi-Fi -->
    <uses-permission android:name="android.permission.INTERNET" />
    <uses-permission android:name="android.permission.ACCESS_NETWORK_STATE" />
    <uses-permission android:name="android.permission.RECEIVE_BOOT_COMPLETED" />

    <application
        android:allowBackup="true"
        android:label="Corsa Cockpit OS"
        android:usesCleartextTraffic="true"
        android:theme="@style/Theme.AppCompat.NoActionBar">

        <activity
            android:name=".MainActivity"
            android:exported="true"
            android:launchMode="singleTask"
            android:screenOrientation="sensorLandscape"
            android:configChanges="orientation|screenSize|keyboardHidden">

            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <!-- Convierte la app en el Launcher Oficial de la TV Box -->
                <category android:name="android.intent.category.HOME" />
                <category android:name="android.intent.category.DEFAULT" />
                <category android:name="android.intent.category.LAUNCHER" />
            </intent-filter>
        </activity>
    </application>
</manifest>
Sketch completo para ESP32 (`corsa_telemetry_esp32.ino`) 
// TELEMETRÍA DIRECTA CHEVROLET CORSA 2011 -> TV BOX ANDROID (USB SERIAL 115200)
#include <Arduino.h>

#define PIN_RPM_PULSE     25  // Señal tacómetro bobina/ECU
#define PIN_VSS_PULSE     26  // Sensor de velocidad VSS caja Corsa
#define PIN_FUEL_ADC      34  // Flotante tanque nafta (54 Litros)
#define PIN_TEMP_ADC      35  // Bulbo temperatura agua
#define PIN_BATT_ADC      32  // Voltaje batería 12V

#define PIN_TURN_L        13  // Señalero Izquierdo
#define PIN_TURN_R        12  // Señalero Derecho
#define PIN_LOW_BEAM      14  // Luz Baja / Iluminación Tablero (Activa Modo Noche)
#define PIN_HIGH_BEAM     27  // Luz Alta
#define PIN_FOG_POS       18  // Luz de Posición / Antiniebla (Activa Modo Noche)
#define PIN_FOOT_BRAKE    33  // Luz de STOP (pedal de freno)
#define PIN_HANDBRAKE     15  // Freno de mano (masa)
#define PIN_CHECK_ENG     4   // Check Engine
#define PIN_OIL_WARN      16  // Bulbo presión de aceite
#define PIN_BATT_WARN     17  // Testigo alternador

volatile unsigned long rpmPulses = 0;
volatile unsigned long vssPulses = 0;
unsigned long lastSampleMs = 0;

void IRAM_ATTR onRpmPulse() { rpmPulses++; }
void IRAM_ATTR onVssPulse() { vssPulses++; }

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RPM_PULSE, INPUT_PULLUP);
  pinMode(PIN_VSS_PULSE, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_RPM_PULSE), onRpmPulse, FALLING);
  attachInterrupt(digitalPinToInterrupt(PIN_VSS_PULSE), onVssPulse, FALLING);

  int inputs[] = {PIN_TURN_L, PIN_TURN_R, PIN_LOW_BEAM, PIN_HIGH_BEAM, PIN_FOG_POS,
                  PIN_FOOT_BRAKE, PIN_HANDBRAKE, PIN_CHECK_ENG, PIN_OIL_WARN, PIN_BATT_WARN};
  for (int p : inputs) pinMode(p, INPUT_PULLUP);
}

void loop() {
  unsigned long now = millis();
  if (now - lastSampleMs >= 25) {
    float dt = (now - lastSampleMs) / 1000.0;
    lastSampleMs = now;

    noInterrupts();
    unsigned long rCount = rpmPulses; rpmPulses = 0;
    unsigned long vCount = vssPulses; vssPulses = 0;
    interrupts();

    int rpm = (int)((rCount / dt) * 30.0);
    int speedKmh = (int)((vCount / dt) * 0.85);
    int fuelPct = constrain(map(analogRead(PIN_FUEL_ADC), 400, 3600, 0, 100), 0, 100);
    int coolantC = constrain(map(analogRead(PIN_TEMP_ADC), 500, 3400, 50, 120), 50, 125);
    float battV = (analogRead(PIN_BATT_ADC) / 4095.0) * 18.5;

    Serial.printf(
      "{\"rpm\":%d,\"spd\":%d,\"fuel\":%d,\"temp\":%d,\"volt\":%.1f,"
      "\"turnL\":%d,\"turnR\":%d,\"lowBeam\":%d,\"highBeam\":%d,\"fog\":%d,"
      "\"brake\":%d,\"handbrake\":%d,\"check\":%d,\"oil\":%d,\"batt\":%d}\n",
      rpm, speedKmh, fuelPct, coolantC, battV,
      !digitalRead(PIN_TURN_L), !digitalRead(PIN_TURN_R),
      !digitalRead(PIN_LOW_BEAM), !digitalRead(PIN_HIGH_BEAM), !digitalRead(PIN_FOG_POS),
      !digitalRead(PIN_FOOT_BRAKE), !digitalRead(PIN_HANDBRAKE),
      !digitalRead(PIN_CHECK_ENG), !digitalRead(PIN_OIL_WARN), !digitalRead(PIN_BATT_WARN)
    );
  }
}
